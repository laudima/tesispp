% ================= Row 2: direct condition ========================
\node[row, text width=2.8cm] (prompt) at (0,-3.7) {%
  \textbf{Prompt} (direct)\\
  1.~background (graph)\\
  2.~given information\\
  3.~question\\
  4.~\emph{Answer with Yes or No only.}};
\node[row, right=0.3cm of prompt.north east, anchor=north west, text width=2.05cm] (model) {%
  \textbf{Model}\\
  4 open-weight\\ models,\\ 1.5B to 7.6B\\
  chat template\\ 1 forward pass\\ no decoding};
\node[row, right=0.3cm of model.north east, anchor=north west, text width=3.0cm] (read) {%
  \textbf{Readout}: logits $\ell_t$ at the first answer token\\[2pt]
  $P(\text{Yes})=\dfrac{S_{\mathcal{Y}}}{S_{\mathcal{Y}}+S_{\mathcal{N}}}$,\;
  $S_{\mathcal{A}}=\textstyle\sum_{t\in\mathcal{A}}e^{\ell_t}$\\[2pt]
  $\mathcal{Y},\mathcal{N}$: case and space\\ variants of Yes, No};
\node[row, right=0.3cm of read.north east, anchor=north west, text width=2.25cm] (ana) {%
  \textbf{Analysis}\\
  Yes iff\\ \hspace*{4pt}$P(\text{Yes})>0.5$,\\ vs.\ ground truth\\[1pt]
  H1: accuracy gap\\
  H1b: $\Delta P(\text{Yes})$\\
  H2: by rung};
